! =====================================================
! EtherChannel LACP S1 <-> S2 (Po1) - 2026-11-20
! Segundo cable agregado: S1 e3/0 <-> S2 e1/3 (el original es S1 e0/2 <-> S2 e0/0)
! Justificacion: el troncal S1-S2 lleva las 4 VLAN y era un punto unico de falla.
! LACP (IEEE 802.3ad, mode active) en vez de PAgP (propietario Cisco).
! =====================================================
! --- S1 ---
configure terminal
interface ethernet3/0
 description EtherChannel a S2 (miembro 2)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
interface range ethernet0/2 , ethernet3/0
 channel-group 1 mode active
interface port-channel 1
 description EtherChannel a S2 (LACP)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
end
write memory

! --- S2 ---
configure terminal
interface ethernet1/3
 description EtherChannel a S1 (miembro 2)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
interface range ethernet0/0 , ethernet1/3
 channel-group 1 mode active
interface port-channel 1
 description EtherChannel a S1 (LACP)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
end
write memory
