! =====================================================
! R2-PAMPLONA - Ruta por defecto, publicacion web, MSS y ACLs - 2026-11-19
! =====================================================
configure terminal
! Anuncio controlado de la ruta por defecto hacia R1 y S3
router rip
 default-information originate
exit
! Publicacion del servidor web hacia la red externa (port forwarding)
ip nat inside source static tcp 192.168.30.2 80 interface ethernet0/3 80
! Ajuste de MSS: los contenedores enviaban segmentos TCP grandes que se perdian
! (la conexion abria pero no llegaban datos). Con MSS 1360 la pagina carga.
interface ethernet0/3
 ip tcp adjust-mss 1360
interface ethernet0/2.30
 ip tcp adjust-mss 1360
exit

! --- ACL 2: proteger el servidor web (VLAN 30) ---
ip access-list extended PROTEGER_WEB
 remark Solo trafico necesario hacia la VLAN 30 (servidor web)
 permit tcp any host 192.168.30.2 eq www
 permit tcp any host 192.168.30.2 eq 443
 permit icmp any 192.168.30.0 0.0.0.255
 permit tcp any 192.168.30.0 0.0.0.255 established
 permit udp host 192.168.20.3 eq domain 192.168.30.0 0.0.0.255
 deny ip any any log
exit
interface ethernet0/2.30
 ip access-group PROTEGER_WEB out
exit

! --- ACL 3: usuarios (VLAN 10) sin acceso a administracion (VLAN 40) ---
ip access-list extended USUARIOS_SIN_ADMIN
 remark VLAN 10 no accede a la VLAN 40, excepto respuestas DHCP
 permit udp host 192.168.10.2 eq bootps any
 deny ip 192.168.10.0 0.0.0.255 192.168.40.0 0.0.0.255 log
 permit ip any any
exit
interface ethernet0/1.10
 ip access-group USUARIOS_SIN_ADMIN in
end
write memory
