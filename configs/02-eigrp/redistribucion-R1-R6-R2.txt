! =====================================================
! Enlace RIP <-> EIGRP y redistribucion - 2026-11-20
! R1-PAMPLONA es el unico punto de redistribucion (ASBR): sin riesgo de bucles.
! La ruta por defecto (RIP desde R2) pasa a EIGRP como D*EX.
! =====================================================
! --- R6-CUCUTA ---
configure terminal
interface ethernet0/0
 description Enlace a R1-PAMPLONA (RIP-EIGRP)
 ip address 192.168.26.1 255.255.255.252
 no shutdown
router eigrp 100
 network 192.168.26.0 0.0.0.3
end
write memory

! --- R1-PAMPLONA ---
configure terminal
interface ethernet0/2
 description Enlace a R6-CUCUTA (EIGRP)
 ip address 192.168.26.2 255.255.255.252
 no shutdown
router eigrp 100
 eigrp router-id 1.1.1.1
 no auto-summary
 network 192.168.26.0 0.0.0.3
 ! seed metric: BW 10000 Kbps, delay 100 (x10 us), reliability 255, load 1, MTU 1500
 redistribute rip metric 10000 100 255 1 1500
router rip
 ! 3 saltos para que RIP prefiera sus rutas internas
 redistribute eigrp 100 metric 3
end
write memory

! --- R2-PAMPLONA: redes EIGRP al NAT ---
configure terminal
ip access-list standard NAT_INSIDE
 permit 192.168.16.0 0.0.7.255
 permit 192.168.26.0 0.0.0.3
 ! (2026-11-20) enlace movido de 192.168.24.0/30 a 192.168.26.0/30 por conflicto con R9-R3
 permit 192.168.50.0 0.0.0.255
end
write memory
